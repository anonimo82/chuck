cmake_minimum_required(VERSION 3.20)
project(ChucKMultimedia LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(MIDIFILE_SOURCE_DIR "" CACHE PATH "Path to midifile source checkout")
find_package(PkgConfig REQUIRED)
pkg_check_modules(MAGICK REQUIRED IMPORTED_TARGET MagickWand)
pkg_check_modules(SVG REQUIRED IMPORTED_TARGET librsvg-2.0>=2.46 cairo)
pkg_check_modules(ASSIMP REQUIRED IMPORTED_TARGET assimp)
pkg_check_modules(SND REQUIRED IMPORTED_TARGET sndfile>=1.2.0)
pkg_check_modules(AV REQUIRED IMPORTED_TARGET libavformat libavcodec libavutil libswscale)
if(NOT EXISTS "${MIDIFILE_SOURCE_DIR}/include/MidiFile.h")
  message(FATAL_ERROR "Set MIDIFILE_SOURCE_DIR to a midifile source checkout")
endif()
function(chugin name source)
  add_library(${name} MODULE ${source} ${ARGN})
  target_include_directories(${name} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../src/core")
  target_compile_definitions(${name} PRIVATE __CHUCK_CHUGIN__)
  set_target_properties(${name} PROPERTIES PREFIX "" SUFFIX ".chug")
  install(TARGETS ${name} LIBRARY DESTINATION chugins RUNTIME DESTINATION chugins)
endfunction()
chugin(RasterIO raster/RasterIO.cpp)
target_link_libraries(RasterIO PRIVATE PkgConfig::MAGICK)
chugin(VectorIO vector/VectorIO.cpp)
target_link_libraries(VectorIO PRIVATE PkgConfig::SVG)
chugin(ModelIO model/ModelIO.cpp)
target_link_libraries(ModelIO PRIVATE PkgConfig::ASSIMP)
chugin(AudioIO audio/AudioIO.cpp)
target_link_libraries(AudioIO PRIVATE PkgConfig::SND)
file(GLOB MIDI_SOURCES CONFIGURE_DEPENDS "${MIDIFILE_SOURCE_DIR}/src/*.cpp")
chugin(MidiFileIO midi/MidiFileIO.cpp ${MIDI_SOURCES})
target_include_directories(MidiFileIO PRIVATE "${MIDIFILE_SOURCE_DIR}/include")
chugin(VideoIO video/VideoIO.cpp)
target_link_libraries(VideoIO PRIVATE PkgConfig::AV)
